cmake_minimum_required(VERSION 3.16)
project(memory_oop_lab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(memory_oop_lab src/main.cpp)
target_include_directories(memory_oop_lab PRIVATE include)
if(MSVC)
  target_compile_options(memory_oop_lab PRIVATE /W4)
else()
  target_compile_options(memory_oop_lab PRIVATE -Wall -Wextra -Wpedantic)
endif()
